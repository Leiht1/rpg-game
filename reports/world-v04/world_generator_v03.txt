class_name WildGenerator
extends RefCounted

## All randomness is coordinate-addressed. No shared RNG or load-order state.
const VERSION := "windrest-biome-0.3.0"
const REGION_SPAN := 8192.0
var region_sites := {}
const CELL := 48
const CHUNK_CELLS := 24
const CHUNK_PX := CELL * CHUNK_CELLS
const TEX_RES := 144
const TEX_STEP := float(CHUNK_PX) / TEX_RES
const PROP_STEP := 144

var world_seed: int
var map_cells: int
var extent: float
var elevation := FastNoiseLite.new()
var moisture := FastNoiseLite.new()
var detail := FastNoiseLite.new()
var phase: float
var warp := FastNoiseLite.new()
var lake_bins := {}
var lakes: Array[Vector4] = []
var spawn := Vector2.ZERO

func _init(seed_value: int = 271828, size_cells: int = 1024) -> void:
	world_seed = seed_value
	map_cells = clampi(size_cells, 128, 1024)
	extent = map_cells * CELL
	moisture.seed=world_seed ^ 117
	moisture.frequency=.00078
	moisture.fractal_octaves=3
	elevation.seed=world_seed ^ 912
	elevation.frequency=.0011
	elevation.fractal_octaves=3
	detail.seed=world_seed ^ 713
	detail.frequency=.006
	detail.fractal_octaves=3
	warp.seed=world_seed ^ 318
	warp.frequency=.00018
	warp.fractal_octaves=3
	spawn=Vector2.ONE*extent*.5
	# Separated, jittered macro sites prevent tiny isolated noise islands.
	for y in range(-2,ceili(extent/REGION_SPAN)+3):
		for x in range(-2,ceili(extent/REGION_SPAN)+3):
			var sx := clampi(x,0,maxi(0,floori(extent/REGION_SPAN)-1))
			var sy := clampi(y,0,maxi(0,floori(extent/REGION_SPAN)-1))
			region_sites[Vector2i(x,y)]=Vector3((sx+.3+random_at(sx,sy,201)*.4)*REGION_SPAN,(sy+.3+random_at(sx,sy,202)*.4)*REGION_SPAN,mini(2,int(random_at(sx,sy,203)*3)))
	var rng=RandomNumberGenerator.new()
	rng.seed=world_seed
	for by in ceili(extent/6144.0):
		for bx in ceili(extent/6144.0):
			for i in rng.randi_range(3,6):
				var p=Vector2(bx,by)*6144+Vector2(rng.randf_range(650,5494),rng.randf_range(550,5594))
				if p.x>extent-400 or p.y>extent-400 or p.distance_to(spawn)<850:continue
				var lake=Vector4(p.x,p.y,rng.randf_range(280,680),rng.randf_range(230,510))
				lakes.append(lake)
				var lo=Vector2i((p-Vector2(lake.z,lake.w)*1.3)/2048)
				var hi=Vector2i((p+Vector2(lake.z,lake.w)*1.3)/2048)
				for y in range(lo.y,hi.y+1):
					for x in range(lo.x,hi.x+1):
						var key=Vector2i(x,y)
						if not lake_bins.has(key):lake_bins[key]=[]
						lake_bins[key].append(lake)

func random_at(x: int, y: int, salt: int = 0) -> float:
	# 31-bit arithmetic stays below signed int64 overflow on all Godot targets.
	var h := (world_seed & 0x7fffffff) ^ ((x * 374761393) & 0x7fffffff) ^ ((y * 668265263) & 0x7fffffff) ^ (salt * 69069)
	h = ((h ^ (h >> 13)) * 1274126177) & 0x7fffffff
	h = (h ^ (h >> 16)) & 0x7fffffff
	return float(h) / 2147483647.0

func sample(p: Vector2, spawn_clearance: bool = true) -> Vector4:
	var q=p+Vector2(warp.get_noise_2d(p.x,p.y),warp.get_noise_2d(p.y+1200,p.x))*240
	var cell := Vector2i(floori(q.x/REGION_SPAN),floori(q.y/REGION_SPAN))
	var distances := Vector3(INF,INF,INF)
	for y in range(cell.y-1,cell.y+2):
		for x in range(cell.x-1,cell.x+2):
			var site: Vector3=region_sites[Vector2i(x,y)]
			var kind := int(site.z)
			distances[kind]=minf(distances[kind],q.distance_to(Vector2(site.x,site.y)))
	var nearest := minf(distances.x,minf(distances.y,distances.z))
	var weights := Vector3.ZERO
	for i in 3: weights[i]=1.0-smoothstep(0.0,700.0,distances[i]-nearest)
	weights/=weights.x+weights.y+weights.z
	var forest=weights.x
	var mountain=weights.y
	var safe=smoothstep(200,470,p.distance_to(spawn)) if spawn_clearance else 1.0
	forest*=safe
	mountain*=safe
	var water=0.0
	for lake in lake_bins.get(Vector2i(q/2048),[]):
		var d=((q-Vector2(lake.x,lake.y))/Vector2(lake.z,lake.w)).length()
		water=maxf(water,1.0-smoothstep(.67,1.18,d))
	return Vector4(forest,mountain,water*safe,1.0-forest-mountain)

func biome(p: Vector2) -> String:
	# A spawn clearing is a detail inside its macro region, not a fourth region.
	var v := sample(p,false)
	if v.z > .56:return "湖泊"
	if v.z > .12:return "水岸"
	if v.y >= v.x and v.y >= v.w: return "山地"
	if v.x >= v.w: return "森林"
	return "草原"

func can_walk(p: Vector2, radius: float = 10.0) -> bool:
	if p.x < radius or p.y < radius or p.x > extent-radius or p.y > extent-radius:
		return false
	for offset in [Vector2.ZERO,Vector2(radius,0),Vector2(-radius,0),Vector2(0,radius),Vector2(0,-radius)]:
		if sample(p + offset).z > .56: return false
	return true

func props_for_chunk(coord: Vector2i) -> Array[Dictionary]:
	var result: Array[Dictionary] = []
	var origin := Vector2(coord) * CHUNK_PX
	var first := Vector2i(origin / PROP_STEP)
	var count := CHUNK_PX / PROP_STEP
	for y in range(first.y, first.y + count):
		for x in range(first.x, first.x + count):
			var p := Vector2((x+.18+random_at(x,y,1)*.64)*PROP_STEP,(y+.18+random_at(x,y,2)*.64)*PROP_STEP)
			if p.x >= extent-64 or p.y >= extent-64 or p.distance_to(spawn) < 480: continue
			var v := sample(p)
			if v.z > .03: continue
			var roll := random_at(x,y,3)
			var cluster := smoothstep(-.4,.35,detail.get_noise_2dv(p))
			var kind := -1
			var scale_value := .8 + random_at(x,y,5)*.35
			if roll < (.025 + v.x*.65) * cluster:
				kind = 0 if random_at(x,y,7)<.7 else 1
			elif roll > .72 and roll < .75 + v.y*.22*cluster:
				kind = 2 if random_at(x,y,7)<.7 else 3
			elif roll > .96:
				kind = 4 + mini(2,int(random_at(x,y,8)*3))
			if kind >= 0:
				result.append({"position":p,"kind":kind,"scale":scale_value,"id":"%d:%d"%[x,y]})
	# A deterministic open test clearing, using the same streaming installation path.
	var samples := [[Vector2(150,-35),0,1.0],[Vector2(-220,-90),1,.95],[Vector2(320,-170),0,1.05],[Vector2(-360,130),0,.9],[Vector2(260,165),1,1.0],[Vector2(-125,120),2,1.0],[Vector2(120,155),3,1.0],[Vector2(390,50),2,.9],[Vector2(-50,-135),4,1.0],[Vector2(205,65),5,1.0],[Vector2(-295,30),6,1.0],[Vector2(65,-195),4,.9]]
	for i in samples.size():
		var p: Vector2=spawn+samples[i][0]
		if Vector2i(floori(p.x/CHUNK_PX),floori(p.y/CHUNK_PX))==coord and can_walk(p,45):
			result.append({"position":p,"kind":samples[i][1],"scale":samples[i][2],"id":"clearing:%d"%i})
	return result

func build_chunk(coord: Vector2i) -> Dictionary:
	var begin := Time.get_ticks_usec()
	var bytes := PackedByteArray()
	bytes.resize((TEX_RES+2)*(TEX_RES+2)*4)
	var index := 0
	var origin := Vector2(coord) * CHUNK_PX
	for y in range(-1,TEX_RES+1):
		for x in range(-1,TEX_RES+1):
			var p := origin + Vector2(x,y)*TEX_STEP
			var v := sample(p)
			bytes[index] = roundi(v.x*255)
			bytes[index+1] = roundi(v.y*255)
			bytes[index+2] = roundi(v.w*255)
			bytes[index+3] = roundi(v.z*255)
			index += 4
	return {"coord":coord,"image":Image.create_from_data(TEX_RES+2,TEX_RES+2,false,Image.FORMAT_RGBA8,bytes),
		"props":props_for_chunk(coord),"build_ms":(Time.get_ticks_usec()-begin)/1000.0}
